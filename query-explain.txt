SELECT
    c.name AS category_name,
    SUM(oi.quantity * oi.price) AS total_revenue
FROM
    categories c
JOIN
    products p ON c.id = p.category_id -> (product have category id = x)
JOIN
    order_items oi ON p.id = oi.product_id
JOIN
    orders o ON oi.order_id = o.id
JOIN
    users u ON o.user_id = u.id
WHERE
    u.region = 'North' -- Filter by region
    AND o.order_date BETWEEN '2024-10-01' AND '2024-12-31' -- Filter by date range
GROUP BY
    c.name
ORDER BY
    total_revenue DESC
LIMIT 10;
